Phase 2 & 3 Combined Design
IPC Message Structure
A single RPMsg endpoint with a msg_type field handles all traffic:

ipc_message.h
    #define IPC_MSG_TYPE_LOG     0x01  // CM4 -> CM7: log output
    #define IPC_MSG_TYPE_CMD     0x02  // CM7 -> CM4: shell command
    #define IPC_MSG_TYPE_RESP    0x03  // CM4 -> CM7: command response

    #define IPC_MAX_PAYLOAD      224   // keeps total msg under 256 bytes

    typedef struct {
        uint8_t  msg_type;
        uint8_t  reserved[3];
        uint32_t timestamp_ms;
        char     payload[IPC_MAX_PAYLOAD];
    } ipc_message_t;
